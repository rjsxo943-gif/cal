cmake_minimum_required(VERSION 3.21)
project(scientific_calculator_cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(calculator_core STATIC src/ScientificCalculator.cpp)
target_include_directories(calculator_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(scientific_calculator main.cpp)
target_link_libraries(scientific_calculator PRIVATE calculator_core)
if(MSVC)
  target_compile_options(calculator_core PRIVATE /utf-8 /W4 /permissive-)
  target_compile_options(scientific_calculator PRIVATE /utf-8 /W4 /permissive-)
else()
  target_compile_options(calculator_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  target_compile_options(scientific_calculator PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(calculator_tests tests/CalculatorTests.cpp)
  target_link_libraries(calculator_tests PRIVATE calculator_core)
  target_include_directories(calculator_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  if(MSVC)
    target_compile_options(calculator_tests PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(calculator_tests PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
  add_test(NAME calculator_tests COMMAND calculator_tests)
endif()
