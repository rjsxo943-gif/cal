cmake_minimum_required(VERSION 3.21)
project(scientific_calculator_cpp VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

function(calculator_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /utf-8 /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
  endif()
endfunction()

add_library(calculator_core STATIC src/ScientificCalculator.cpp)
target_include_directories(calculator_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
calculator_warnings(calculator_core)

if(WIN32)
  add_executable(scientific_calculator WIN32 win32/ScientificCalculatorWin32.cpp)
  target_link_libraries(scientific_calculator PRIVATE calculator_core Comctl32)
  target_compile_definitions(scientific_calculator PRIVATE UNICODE _UNICODE)
  calculator_warnings(scientific_calculator)

  add_executable(scientific_calculator_cli main.cpp)
  target_link_libraries(scientific_calculator_cli PRIVATE calculator_core)
  calculator_warnings(scientific_calculator_cli)
else()
  add_executable(scientific_calculator main.cpp)
  target_link_libraries(scientific_calculator PRIVATE calculator_core)
  calculator_warnings(scientific_calculator)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(calculator_tests tests/CalculatorTests.cpp)
  target_link_libraries(calculator_tests PRIVATE calculator_core)
  target_include_directories(calculator_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
  calculator_warnings(calculator_tests)
  add_test(NAME calculator_tests COMMAND calculator_tests)
endif()
